from flask_mail import Mail, Message
mail = Mail(app)

def send_verification_email(to_email, token):
    link = f"https://yourdomain.com/verify/{token}"
    msg = Message("E-posta doğrulama", sender=app.config['MAIL_USERNAME'], recipients=[to_email])
    msg.body = f"Lütfen e-postanızı doğrulayın: {link}"
    mail.send(msg)
